server {
    listen 80;
    server_name webhook-agent.duckdns.org;
    
    # Redirección a HTTPS
    location / {
        return 301 https://$server_name$request_uri;
    }
    
    # Excepción para Let's Encrypt (debe ir ANTES que la redirección)
    location /.well-known/acme-challenge/ {
        root /var/www/html;
    }
}

server {
    listen 443 ssl;
    server_name webhook-agent.duckdns.org;
    ssl_certificate /etc/letsencrypt/live/webhook-agent.duckdns.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/webhook-agent.duckdns.org/privkey.pem;
    
    # Raíz - Flask
    location / {
        proxy_pass http://127.0.0.1:3333/;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    
    # Endpoint específico
    location /agentmail {
        proxy_pass http://127.0.0.1:3333/agentmail;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
